# =====================================================================
# VISUAL 1.1 — MAPA COROPLÉTICO DO FATURAMENTO POR ESTADO
# =====================================================================
# Arraste para o visual Python: estado + total_pedido (fact_vendas)

import matplotlib.pyplot as plt
import geopandas as gpd
import numpy as np

col_estado = next((c for c in dataset.columns if 'estado' in c.lower() or 'uf' in c.lower() or 'sigla' in c.lower()), None)
col_valor  = next((c for c in dataset.columns if 'total' in c.lower() or 'faturamento' in c.lower() or 'pedido' in c.lower()), None)

if not col_estado or not col_valor:
    raise KeyError(f"Colunas: {list(dataset.columns)}. Arraste 'estado' e 'total_pedido'.")

gdf = gpd.read_file(r'C:\dev\projetoInt\bi_ecommerce\data\brazil_states.geojson')
df_g = dataset.groupby(col_estado)[col_valor].sum().reset_index()
gdf = gdf.merge(df_g, left_on='sigla', right_on=col_estado, how='left')
gdf[col_valor] = gdf[col_valor].fillna(0)

fig, ax = plt.subplots(figsize=(7, 7), facecolor='#1B1B1B')
ax.set_facecolor('#1B1B1B')

gdf.plot(column=col_valor, cmap='Reds', linewidth=0.4, ax=ax,
         edgecolor='#2D2D2D', legend=True,
         legend_kwds={'label': 'Faturamento (R$)', 'orientation': 'horizontal',
                      'pad': 0.01, 'shrink': 0.55})

fig.axes[-1].tick_params(colors='#AAAAAA', labelsize=7)
fig.axes[-1].xaxis.label.set_color('#AAAAAA')
fig.axes[-1].xaxis.label.set_fontsize(8)
ax.axis('off')
plt.title('Faturamento por Estado', fontsize=11, color='#FFFFFF', pad=6, weight='bold', family='Segoe UI')
plt.tight_layout(pad=0.5)
plt.show()


# =====================================================================
# VISUAL 1.2 — RANKING DE VENDEDORES (FINO E ESTICÁVEL)
# =====================================================================
# Arraste para o visual Python: nome (dim_vendedores) + total_pedido (fact_vendas)

import matplotlib.pyplot as plt
import numpy as np

col_nome  = next((c for c in dataset.columns if 'nome' in c.lower() or 'vendedor' in c.lower()), None)
col_valor = next((c for c in dataset.columns if 'total' in c.lower() or 'faturamento' in c.lower() or 'pedido' in c.lower()), None)

if not col_nome or not col_valor:
    raise KeyError(f"Colunas: {list(dataset.columns)}. Arraste 'nome' e 'total_pedido'.")

df = dataset.groupby(col_nome)[col_valor].sum().reset_index().sort_values(col_valor, ascending=True)

fig, ax = plt.subplots(figsize=(14, 4.5), facecolor='#1B1B1B')
ax.set_facecolor('#1B1B1B')
plt.subplots_adjust(left=0.22, right=0.92, top=0.88, bottom=0.12)

bars = ax.barh(df[col_nome], df[col_valor], color='#4A90D9', height=0.55, zorder=3)
ax.barh(df[col_nome], df[col_valor], color='#4A90D9', height=0.55, alpha=0.3, zorder=2)

ax.set_title('Ranking de Vendedores por Faturamento', fontsize=10, color='#FFFFFF',
             pad=8, weight='bold', family='Segoe UI')
ax.tick_params(colors='#AAAAAA', labelsize=8.5)
ax.spines['top'].set_visible(False)
ax.spines['right'].set_visible(False)
ax.spines['left'].set_color('#3A3A3A')
ax.spines['bottom'].set_color('#3A3A3A')
ax.grid(axis='x', color='#3A3A3A', linestyle='--', alpha=0.5, zorder=0)

for bar in bars:
    w = bar.get_width()
    ax.text(w + df[col_valor].max() * 0.015, bar.get_y() + bar.get_height()/2,
            f'R$ {w/1e6:.1f}M', va='center', color='#CCCCCC', fontsize=8, weight='bold', family='Segoe UI')

plt.show()

# =====================================================================
# VISUAL 1.4 — EVOLUÇÃO MENSAL DO FATURAMENTO (LINHA + ÁREA)
# =====================================================================
# Arraste: mes_num + nome_mes (dim_calendario) + total_pedido (fact_vendas)

import matplotlib.pyplot as plt
import numpy as np

col_mes_num  = next((c for c in dataset.columns if 'mes_num' in c.lower()), None)
col_mes_nome = next((c for c in dataset.columns if 'nome_mes' in c.lower() or 'mes_nome' in c.lower()), None)
col_valor    = next((c for c in dataset.columns if 'total' in c.lower() or 'faturamento' in c.lower() or 'pedido' in c.lower()), None)

if not col_mes_num or not col_mes_nome or not col_valor:
    raise KeyError(f"Colunas: {list(dataset.columns)}. Arraste mes_num, nome_mes e total_pedido.")

df = dataset.groupby([col_mes_num, col_mes_nome])[col_valor].sum().reset_index().sort_values(col_mes_num)

fig, ax = plt.subplots(figsize=(14, 3.8), facecolor='#1B1B1B')
ax.set_facecolor('#1B1B1B')
plt.subplots_adjust(left=0.08, right=0.97, top=0.85, bottom=0.15)

ax.fill_between(range(len(df)), df[col_valor], color='#4A90D9', alpha=0.15)
ax.plot(range(len(df)), df[col_valor], color='#4A90D9', linewidth=2.2, marker='o',
        markersize=6, markerfacecolor='#1B1B1B', markeredgecolor='#4A90D9', markeredgewidth=2)

ax.set_xticks(range(len(df)))
ax.set_xticklabels(df[col_mes_nome], rotation=0)
ax.set_title('Evolução Mensal do Faturamento', fontsize=10, color='#FFFFFF',
             pad=8, weight='bold', family='Segoe UI')
ax.tick_params(colors='#AAAAAA', labelsize=8)
ax.spines['top'].set_visible(False)
ax.spines['right'].set_visible(False)
ax.spines['left'].set_color('#3A3A3A')
ax.spines['bottom'].set_color('#3A3A3A')
ax.grid(color='#3A3A3A', linestyle='--', alpha=0.5)
ax.yaxis.set_major_formatter(plt.FuncFormatter(lambda x, _: f'R$ {x/1e6:.1f}M'))

for i, v in enumerate(df[col_valor]):
    ax.annotate(f'R$ {v/1e6:.1f}M', (i, v), textcoords='offset points', xytext=(0, 12),
                ha='center', color='#CCCCCC', fontsize=7.5, weight='bold', family='Segoe UI')

plt.show()
